CREATE DATABASE IF NOT EXISTS respondapp_basico
    CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE respondapp_basico;
SET NAMES utf8mb4;

-- 1. Usuarios que pueden ingresar al sistema.
CREATE TABLE usuario (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre_completo VARCHAR(100) NOT NULL,
    nombre_usuario VARCHAR(40) NOT NULL UNIQUE,
    fecha_nacimiento DATE NOT NULL,
    email VARCHAR(150) NOT NULL,
    clave VARCHAR(150) NOT NULL,
    rol ENUM('ADMINISTRADOR', 'JUGADOR') NOT NULL
) ENGINE=InnoDB;

-- 2. Temas de las preguntas
CREATE TABLE categoria (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(60) NOT NULL UNIQUE,
    descripcion VARCHAR(250) NOT NULL
) ENGINE=InnoDB;

-- 3. Cada pregunta pertenece a una categoria
CREATE TABLE pregunta (
    id_pregunta INT AUTO_INCREMENT PRIMARY KEY,
    id_categoria INT NOT NULL,
    texto VARCHAR(500) NOT NULL,
    respuesta_correcta VARCHAR(250) NOT NULL,
    dificultad ENUM('facil', 'medio', 'dificil') NOT NULL,
    FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria)
) ENGINE=InnoDB;

-- 4. Las cuatro respuestas que se muestran para una pregunta
-- id_pregunta puede ser NULL para una opcion de reserva
CREATE TABLE opcion (
    id_opcion INT AUTO_INCREMENT PRIMARY KEY,
    id_pregunta INT,
    numero TINYINT NOT NULL,
    texto VARCHAR(250) NOT NULL,
    UNIQUE (id_pregunta, numero),
    FOREIGN KEY (id_pregunta) REFERENCES pregunta(id_pregunta),
    CHECK (numero BETWEEN 1 AND 4)
) ENGINE=InnoDB;

-- 5. Una partida entre dos jugadores
CREATE TABLE partida (
    id_partida INT AUTO_INCREMENT PRIMARY KEY,
    id_jugador1 INT NOT NULL,
    id_jugador2 INT NOT NULL,
    id_jugador_activo INT NOT NULL,
    fecha DATE NOT NULL,
    hora_comienzo TIME NOT NULL,
    hora_actual TIME NOT NULL,
    hora_finalizacion TIME,
    inicio DATETIME NOT NULL,
    fin DATETIME,
    estado ENUM('en curso', 'finalizada') NOT NULL,
    total_rondas INT NOT NULL,
    FOREIGN KEY (id_jugador1) REFERENCES usuario(id_usuario),
    FOREIGN KEY (id_jugador2) REFERENCES usuario(id_usuario),
    FOREIGN KEY (id_jugador_activo) REFERENCES usuario(id_usuario),
    CHECK (id_jugador1 <> id_jugador2),
    CHECK (id_jugador_activo IN (id_jugador1, id_jugador2)),
    CHECK (total_rondas BETWEEN 2 AND 20 AND MOD(total_rondas, 2) = 0)
) ENGINE=InnoDB;

-- 6. Respuestas guardadas en cada turno de una partida
CREATE TABLE ronda (
    id_partida INT NOT NULL,
    numero INT NOT NULL,
    id_jugador INT NOT NULL,
    id_pregunta INT NOT NULL,
    id_categoria INT NOT NULL,
    respuesta_dada VARCHAR(250) NOT NULL,
    respuesta_correcta VARCHAR(250) NOT NULL,
    acierto ENUM('SI', 'NO') NOT NULL,
    dificultad ENUM('facil', 'medio', 'dificil') NOT NULL,
    puntos INT NOT NULL,
    PRIMARY KEY (id_partida, numero),
    UNIQUE (id_partida, id_pregunta),
    FOREIGN KEY (id_partida) REFERENCES partida(id_partida),
    FOREIGN KEY (id_jugador) REFERENCES usuario(id_usuario),
    FOREIGN KEY (id_pregunta) REFERENCES pregunta(id_pregunta),
    FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria),
    CHECK (numero > 0),
    CHECK ((acierto = 'SI' AND puntos = 10) OR (acierto = 'NO' AND puntos = 0))
) ENGINE=InnoDB;

-- 7. Puntos y posicion de cada jugador en cada categoria
CREATE TABLE ranking (
    id_ranking INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_categoria INT NOT NULL,
    puntos INT NOT NULL DEFAULT 0,
    posicion INT NOT NULL DEFAULT 1,
    UNIQUE (id_usuario, id_categoria),
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria),
    CHECK (puntos >= 0 AND posicion > 0)
) ENGINE=InnoDB;

-- CONSULTAS DEL PROYECTO
USE respondapp_basico;

-- 1. Cantidad de preguntas dificiles sobre Deportes
SELECT COUNT(*) AS preguntas_dificiles_de_deportes
FROM pregunta p JOIN categoria c ON c.id_categoria=p.id_categoria
WHERE LOWER(c.nombre)='deportes' AND p.dificultad='dificil';

-- 2. Minutos de una partida finalizada. Incluye cambios de dia
SET @partida=1;
SELECT id_partida, ROUND(TIMESTAMPDIFF(SECOND,inicio,fin)/60.0,2) AS minutos
FROM partida WHERE id_partida=@partida AND estado='finalizada';

-- 3. Puntos historicos del jugador (incluye partidas en curso)
SET @jugador=2;
SELECT u.nombre_usuario,COALESCE(SUM(r.puntos),0) AS puntos_historial
FROM usuario u LEFT JOIN ronda r ON r.id_jugador=u.id_usuario
WHERE u.id_usuario=@jugador AND u.rol='JUGADOR'
GROUP BY u.id_usuario,u.nombre_usuario;

-- 4. La consigna dice 'rondas con partidas faciles'. Como la dificultad
-- pertenece a la pregunta, contamos rondas con pregunta facil
SELECT COUNT(*) AS rondas_con_preguntas_faciles FROM ronda WHERE dificultad='facil';

-- 5. Periodos mensuales ordenados por cantidad de partidas iniciadas
-- El primer resultado indica el periodo de mayor frecuencia; se ven los empates
SELECT DATE_FORMAT(inicio,'%Y-%m') AS periodo,COUNT(*) AS partidas
FROM partida GROUP BY DATE_FORMAT(inicio,'%Y-%m') ORDER BY partidas DESC,periodo;

-- 6. Participaciones por jugador, incluyendo ambos lugares de la partida
SELECT u.nombre_usuario,COUNT(p.id_partida) AS juegos
FROM usuario u LEFT JOIN partida p ON u.id_usuario=p.id_jugador1 OR u.id_usuario=p.id_jugador2
WHERE u.rol='JUGADOR'
GROUP BY u.id_usuario,u.nombre_usuario ORDER BY juegos DESC,u.nombre_usuario;
